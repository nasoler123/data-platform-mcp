def _base_fact_monthly_rollup_select(catalog: Mapping[str, Any], read_parquet_sql: str) -> str:
    """SELECT that re-aggregates a (now DAY-grain) base_fact parquet back to MONTHLY.

    The base_fact is additive-only (sum/count), so summing the daily rows per
    month × dims × driver reproduces the historical monthly base_fact bit-for-bit.
    The pivot engine reads the daily rows directly (for day/week); the FREE-SQL
    surface keeps its historical monthly ``period_start`` contract so existing
    ``GROUP BY period_start`` SQL views don't silently flip from monthly to daily
    after the grain migration (Codex P2). No-op on an already-monthly (legacy)
    base_fact: ``DATE_TRUNC('month', ...)`` of a month-start is the same month.
    """
    month_expr = f"DATE_TRUNC('month', CAST({_quote_ident('period_start')} AS DATE))"
    dims = [dim for dim in _columns_by_role(catalog, "dimension")]
    key_cols = [*dims, "driver_id"]
    measures = [col for col in _BASE_FACT_VALUE_COLUMNS if col != "driver_id"]
    select_cols = [f"{month_expr}::DATE AS {_quote_ident('period_start')}"]
    group_cols = [month_expr]
    for col in key_cols:
        select_cols.append(_quote_ident(col))
        group_cols.append(_quote_ident(col))
    for col in measures:
        select_cols.append(f"SUM({_quote_ident(col)}) AS {_quote_ident(col)}")
    return (
        f"SELECT {', '.join(select_cols)} FROM {read_parquet_sql} "
        f"GROUP BY {', '.join(group_cols)}"
    )
