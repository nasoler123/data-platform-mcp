def _strict_account_dataset_ids(user_id: str, account_id: str) -> Optional[set[str]]:
    """Canonical ids whose account ownership is proven by scoped stores."""

    try:
        from api.memory_semantic_graph_providers import ChalsDatasetsProvider

        records = ChalsDatasetsProvider().list_datasets(
            account_id,
            user_id=user_id,
        )
    except Exception:  # noqa: BLE001 - the public boundary fails closed
        logger.exception(
            "chalsai_mcp.analyst.strict_inventory_failed user=%s account=%s",
            user_id,
            account_id,
        )
        return None
    visible: set[str] = set()
    for record in records:
        if not isinstance(record, Mapping):
            continue
        for field in ("id", "dataset_id", "dataset_key"):
            value = str(record.get(field) or "").strip()
            if value:
                visible.add(value)
    return visible
