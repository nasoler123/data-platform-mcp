def validate_bi_sql(sql: Any) -> str:
    """Validate one read-only SELECT/WITH statement; return it normalized:
    comments stripped and trailing ``;`` removed, so the executor can safely
    wrap it as a subquery. Raises ``BiSqlRejected``."""
    raw = str(sql or "").strip()
    if not raw:
        raise BiSqlRejected("BI_SQL_EMPTY", "La consulta SQL está vacía.")
    stripped = _strip_comments(raw).strip()
    if not stripped:
        raise BiSqlRejected("BI_SQL_EMPTY", "La consulta SQL solo contiene comentarios.")
    blanked = _blank_quoted_regions(stripped)
    semi = blanked.find(";")
    if semi != -1 and blanked[semi:].strip(" ;\t\r\n"):
        raise BiSqlRejected(
            "BI_SQL_MULTI_STATEMENT",
            "Solo se permite una única sentencia SELECT/WITH (hay contenido tras un ';').",
        )
    body = stripped.rstrip().rstrip(";").rstrip()
    blanked_body = blanked.rstrip().rstrip(";").rstrip()
    match = _FIRST_WORD_RE.match(blanked_body)
    first = (match.group(1).lower() if match else "")
    # DuckDB's FROM-first syntax (``FROM t SELECT ...``) is read-only too.
    if first not in {"select", "with", "from"}:
        raise BiSqlRejected(
            "BI_SQL_NOT_A_QUERY",
            "Solo consultas de lectura: la sentencia debe empezar por SELECT o WITH.",
        )
    for kw in _DENIED_KEYWORDS:
        if _word_in(blanked_body, kw):
            raise BiSqlRejected(
                "BI_SQL_FORBIDDEN_KEYWORD",
                f"Palabra clave no permitida en la consulta: {kw.upper()}. "
                'Si es el nombre de una columna, escríbelo entre comillas dobles ("...").',
            )
    for fn in _DENIED_FUNCTIONS:
        if _word_in(blanked_body, fn):
            raise BiSqlRejected(
                "BI_SQL_FORBIDDEN_FUNCTION",
                f"Función no permitida: {fn}. Consulta únicamente las tablas de material del workspace.",
            )
    # Quoted relation source (DuckDB's implicit ``FROM 'file.csv'`` → read_csv). Scan
    # the comment-stripped statement BEFORE blanking — the literal is the evidence.
    if _FILE_RELATION_RE.search(stripped):
        raise BiSqlRejected(
            "BI_SQL_FORBIDDEN_FILE_SOURCE",
            "Origen de tabla no permitido: una ruta entre comillas tras FROM/JOIN lee un "
            "fichero local. Consulta únicamente las tablas de material (raw_<dataset>).",
        )
    return body
