def _input_schema(tool: Any) -> dict[str, Any]:
    """Closed, typed JSON Schema for the public MCP contract.

    ``tools.yaml`` remains the source for the in-product modeler. The MCP edge is
    a separately versioned public API: it cannot expose every legacy alias a
    handler happens to accept, because undeclared aliases are precisely how a
    caller can bypass account checks or trigger a silent no-op. Each reviewed
    tool therefore has one canonical wire shape and rejects extra properties.
    """
    schema = _MCP_INPUT_SCHEMAS.get(str(getattr(tool, "name", "") or ""))
    if schema is None:
        raise AnalystDoctrineError(
            f"No reviewed MCP input schema for allowlisted tool {getattr(tool, 'name', '(unknown)')}"
        )
    projected = deepcopy(schema)
    if str(getattr(tool, "name", "") or "") == "bi_analyze":
        # Some hosts render anyOf as a union and retain only each branch's
        # properties. Mode-only branches then hide dataset_keys/metrics and
        # their generated calls reach Core without the requested intent.
        # Make each branch self-contained without changing accepted payloads.
        for branch in projected["anyOf"]:
            branch["type"] = "object"
            branch["properties"] = {
                **deepcopy(projected["properties"]), **branch.get("properties", {})
            }
            branch["required"] = list(dict.fromkeys(
                projected.get("required", []) + branch.get("required", [])
            ))
            branch["additionalProperties"] = False
    return projected
