````sql
{{ config(meta={'chals_tabla': 'pedidos'}) }}
-- Cada control devuelve las filas que FALLAN. Cero filas = verde.
--
-- Los controles de identidad son la lección de [client]: un número puede ser
-- plausible, sumar bien por su lado y significar otra cosa. Lo único que lo caza
-- es exigir que las piezas cuadren entre sí, FILA A FILA. Se excluyen los pedidos
-- editados tras la venta, cuyos importes vigentes ya no son los originales.
WITH validos AS (
  SELECT p.*, NOT p.cancelado AND NOT p.es_prueba
    {%- if chals_origen_presente(['order_line']) %}
    AND IFNULL(l.con_producto, TRUE)
    {%- endif %} AS valido
  FROM {{ ref('pedidos') }} AS p
  {%- if chals_origen_presente(['order_line']) %}
  LEFT JOIN (
    SELECT CAST(order_id AS STRING) AS pedido_id,
           LOGICAL_OR(NOT IFNULL(is_gift_card, FALSE)) AS con_producto
    FROM {{ source('shopify', 'order_line') }}
    GROUP BY order_id
  ) AS l USING (pedido_id)
  {%- endif %}
)
SELECT 'pedido_id duplicado' AS control, CAST(pedido_id AS STRING) AS detalle
FROM {{ ref('pedidos') }}
GROUP BY pedido_id HAVING COUNT(*) > 1
UNION ALL
SELECT 'pedido_id nulo', 'fila sin clave'
FROM {{ ref('pedidos') }} WHERE pedido_id IS NULL
UNION ALL
SELECT 'fecha de pedido nula o futura', IFNULL(CAST(fecha_pedido AS STRING), 'NULL')
FROM {{ ref('pedidos') }}
WHERE fecha_pedido IS NULL OR fecha_pedido > CURRENT_DATE({{ chals_zona_tienda() }})
UNION ALL

-- Identidad del pedido: lo cobrado es producto + envío cobrado + propina +
-- aranceles + impuestos (solo cuando NO van incluidos en el precio).
SELECT 'identidad del total rota',
       FORMAT('%s: total=%t productos=%t envio_cobrado=%t impuestos=%t incluidos=%t propina=%t',
              numero_pedido, importe_total, importe_productos, envio_cobrado, impuestos,
              impuestos_incluidos, propina)
FROM {{ ref('pedidos') }}
WHERE NOT editado
  AND ABS(importe_total - (importe_productos + envio_cobrado + propina + aranceles
                           + IF(impuestos_incluidos, 0, impuestos))) > 0.005
UNION ALL
````
