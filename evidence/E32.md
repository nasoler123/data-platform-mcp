````sql
-- Grano: un día, desde la primera hasta la última fecha con pedidos o anuncios.
--
-- Sin dimensiones a propósito: aloja solo las métricas combinadas que cruzan la
-- tienda con la publicidad (MER, CAC combinado, POAS…), que se pueden cortar por
-- fecha y por nada más. Cortar la inversión de Meta por producto o las ventas
-- por campaña sería inventar una atribución. Antes se llamaba `calendario`.
--
-- `inversion_en_tienda` (3.3.0): la inversión del día (moneda de referencia,
-- como `anuncios.inversion`) que vende en la tienda, la que usan las
-- combinadas. Cuenta la de las filas de `anuncios` con canal 'Tienda online'
-- (Google, Meta y las campañas web de TikTok) y la de un canal de marketplace
-- ('TikTok Shop', 'Amazon') solo si ese canal aparece en los pedidos de Shopify
-- de la copia (sin pruebas): si los pedidos de TikTok Shop o de Amazon no entran
-- en la tienda, su inversión no vende en ella y se mide con sus propias
-- métricas (ROAS declarado, ACOS). Un día sin esa inversión, o cuyas cuentas
-- están todas en otra moneda, queda vacío: sin dato no es cero. Con solo Google
-- y Meta es la inversión total de cada día.
WITH limites AS (
  SELECT MIN(dia) AS desde, MAX(dia) AS hasta
  FROM (
    SELECT fecha_pedido AS dia FROM {{ ref('pedidos') }}
    UNION ALL
    SELECT fecha AS dia FROM {{ ref('anuncios') }}
  )
),

canales_tienda AS (
  SELECT DISTINCT canal FROM {{ ref('pedidos') }}
  WHERE NOT es_prueba AND canal IN ('TikTok Shop', 'Amazon')
),

inversion AS (
  SELECT fecha, SUM(inversion) AS inversion_en_tienda
  FROM {{ ref('anuncios') }}
  WHERE canal_venta = 'Tienda online' OR canal_venta IN (SELECT canal FROM canales_tienda)
  GROUP BY fecha
)

SELECT dias.fecha, inversion.inversion_en_tienda, '{{ var("chals_user_id") }}' AS user_id
FROM (SELECT fecha FROM limites, UNNEST(GENERATE_DATE_ARRAY(desde, hasta)) AS fecha) AS dias
LEFT JOIN inversion ON inversion.fecha = dias.fecha
````
